cmake_minimum_required(VERSION 3.13)
set(CMAKE_C_STANDARD 11)
set(PICO_BOARD pico2 CACHE STRING "Board type")
set(PICO_PLATFORM rp2350)
include(pico_sdk_import.cmake)
project(ragchewberry_display C CXX ASM)
pico_sdk_init()
add_executable(ragchewberry_display main.c oled_ui.c)
target_link_libraries(ragchewberry_display PRIVATE pico_stdlib hardware_uart hardware_i2c hardware_dma hardware_irq hardware_adc hardware_flash pico_flash)
pico_enable_stdio_usb(ragchewberry_display 1)
pico_enable_stdio_uart(ragchewberry_display 0)
pico_set_program_name(ragchewberry_display "WP3DN RagchewBerry Display")
pico_set_program_version(ragchewberry_display "0.2-uart-irq")
pico_add_extra_outputs(ragchewberry_display)
